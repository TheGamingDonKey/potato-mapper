cmake_minimum_required(VERSION 3.24)
project(PotatoMapper VERSION 0.8.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Widgets OpenGLWidgets Multimedia Network)
qt_add_executable(PotatoMapper WIN32
    src/main.cpp src/model.h src/model.cpp src/canvas.h src/canvas.cpp src/branding.h src/updates.h src/updates.cpp src/ui.h src/style.h src/patterns.h
    src/dynamic.h src/dynamic.cpp src/projects.h src/projects.cpp)
set_target_properties(PotatoMapper PROPERTIES OUTPUT_NAME PotatoMapperApp)
target_compile_definitions(PotatoMapper PRIVATE POTATO_VERSION="${PROJECT_VERSION}")
target_link_libraries(PotatoMapper PRIVATE Qt6::Widgets Qt6::OpenGLWidgets Qt6::Multimedia Qt6::Network)
qt_add_resources(PotatoMapper branding PREFIX "/" FILES assets/potato-mapper.png assets/patterns.frag packaging/extract-update.ps1)
if(WIN32)
    target_sources(PotatoMapper PRIVATE assets/potato-mapper.rc)
    add_executable(PotatoLauncher WIN32 src/launcher.cpp assets/potato-mapper.rc)
    set_target_properties(PotatoLauncher PROPERTIES OUTPUT_NAME PotatoMapper MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_libraries(PotatoLauncher PRIVATE shell32 comctl32 bcrypt)
    target_compile_definitions(PotatoLauncher PRIVATE UNICODE _UNICODE)
    add_executable(PotatoLauncherRefresh WIN32 src/launcher-refresh.cpp)
    set_target_properties(PotatoLauncherRefresh PROPERTIES OUTPUT_NAME PotatoLauncherRefresh AUTOMOC OFF MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_libraries(PotatoLauncherRefresh PRIVATE shell32 bcrypt)
    target_compile_definitions(PotatoLauncherRefresh PRIVATE UNICODE _UNICODE)
endif()
if(MSVC)
    target_compile_options(PotatoMapper PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(PotatoLauncher PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(PotatoLauncherRefresh PRIVATE /W4 /permissive- /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(PotatoDynamicChecks tests/dynamic.cpp src/dynamic.cpp)
    target_include_directories(PotatoDynamicChecks PRIVATE src)
    target_link_libraries(PotatoDynamicChecks PRIVATE Qt6::Gui)
    add_executable(PotatoProjectChecks tests/projects.cpp src/projects.cpp)
    target_include_directories(PotatoProjectChecks PRIVATE src)
    target_link_libraries(PotatoProjectChecks PRIVATE Qt6::Core)
    add_executable(PotatoUpdateChecks tests/updates.cpp src/updates.cpp src/updates.h)
    target_include_directories(PotatoUpdateChecks PRIVATE src)
    target_link_libraries(PotatoUpdateChecks PRIVATE Qt6::Widgets Qt6::Network)
    qt_add_resources(PotatoUpdateChecks update_script PREFIX "/" FILES packaging/extract-update.ps1)
    if(MSVC)
        target_compile_options(PotatoUpdateChecks PRIVATE /W4 /permissive- /utf-8)
    endif()
endif()
