cmake_minimum_required(VERSION 3.24)
project(PotatoMapper VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Widgets OpenGLWidgets Multimedia)
qt_add_executable(PotatoMapper WIN32
    src/main.cpp src/model.h src/model.cpp src/canvas.h src/canvas.cpp)
target_link_libraries(PotatoMapper PRIVATE Qt6::Widgets Qt6::OpenGLWidgets Qt6::Multimedia)
qt_add_resources(PotatoMapper branding PREFIX "/" FILES assets/potato-mapper.png)
if(WIN32)
    target_sources(PotatoMapper PRIVATE assets/potato-mapper.rc)
endif()
if(MSVC)
    target_compile_options(PotatoMapper PRIVATE /W4 /permissive- /utf-8)
endif()
